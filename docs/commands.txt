awk '{ print }' file.txt
    Print every line in full.

awk '{ print $1 }' file.txt
    Print the first field of each line.

awk '{ print $2 }' file.txt
    Print the second field of each line.

awk '{ print $1, $2 }' file.txt
    Print the first and second fields, space-separated.

awk '{ print $1, $2, $3 }' file.txt
    Print the first three fields.

awk '{ print NF }' file.txt
    Print the number of fields on each line.

awk '{ print NR }' file.txt
    Print the record (line) number of each line.

awk '{ print $NF }' file.txt
    Print the last field of each line.

awk '{ print NR, $1 }' file.txt
    Print the line number followed by the first field.

awk '{ print "text", $1 }' file.txt
    Print a string literal followed by the first field.

awk '{ print 42 }' file.txt
    Print a numeric literal on every line.

awk '{ print $1; print $2 }' file.txt
    Print the first field, then the second field on a new line.

# --- BEGIN / END blocks ---

awk 'BEGIN { print "Start" } { print } END { print "Done" }' file.txt
    Print a header before any input, each line, and a footer after all input.

awk 'BEGINFILE { print "Reading", FILENAME } { print } ENDFILE { print "Finished" }' file.txt
    Run actions at the start and end of each input file.

# --- Regex patterns ---

awk '/Sam/ { print }' file.txt
    Print lines matching the regex /Sam/.

awk '/^S/ { print }' file.txt
    Print lines starting with "S".

awk '/a$/ { print }' file.txt
    Print lines ending with "a".

awk '!/error/ { print }' file.txt
    Print lines that do NOT match /error/.

awk '$1 ~ /^S/ { print }' file.txt
    Print lines where the first field matches /^S/.

awk '$1 !~ /Sam/ { print }' file.txt
    Print lines where the first field does NOT match /Sam/.

# --- Comparison operators ---

awk '$2 > 70 { print $1 }' file.txt
    Print the first field when the second field is greater than 70.

awk '$2 >= 50 { print $1 }' file.txt
    Print the first field when the second field is at least 50.

awk '$2 < 50 { print $1 }' file.txt
    Print the first field when the second field is less than 50.

awk '$2 <= 50 { print $1 }' file.txt
    Print the first field when the second field is at most 50.

awk '$1 == "Sam" { print }' file.txt
    Print lines where the first field equals "Sam".

awk '$1 != "Sam" { print }' file.txt
    Print lines where the first field is not "Sam".

# --- Logical operators ---

awk '$1 > 10 && $2 < 20 { print }' file.txt
    Print lines where both conditions are true (AND).

awk '$1 == "A" || $1 == "B" { print }' file.txt
    Print lines where either condition is true (OR).

# --- Ternary operator ---

awk '{ print $2 > 50 ? "Pass" : "Fail" }' file.txt
    Print "Pass" or "Fail" based on the second field's value.

# --- Parenthesized expressions ---

awk '($1 > 10 && $2 < 20) { print }' file.txt
    Use parentheses to group conditions.

# --- Range patterns ---

awk '/start/,/end/ { print }' file.txt
    Print lines from the first match of /start/ through the first match of /end/.

awk '$1 == "start", $1 == "end" { print }' file.txt
    Print lines from where $1 equals "start" through where $1 equals "end".

# --- Variables and assignment ---

awk '{ x = $2; print x }' file.txt
    Assign the second field to variable x and print it.

awk '{ total += $2 } END { print total }' file.txt
    Sum up the second field across all lines, then print the total.

awk '{ count++ } END { print count }' file.txt
    Count the number of lines processed using the increment operator.

awk 'BEGIN { x = 5; print ++x }'
    Prefix increment: increment x first, then print (prints 6).

awk 'BEGIN { x = 5; print x++; print x }'
    Postfix increment: print x first (5), then increment (now 6).

awk 'BEGIN { x = 5; print --x }'
    Prefix decrement: decrement x first, then print (prints 4).

awk 'BEGIN { x = 5; x--; print x }'
    Postfix decrement: print x first (5), then decrement (now 4).

# --- Arithmetic operators ---

awk '{ print $2 + 10 }' file.txt
    Add 10 to the second field of each line.

awk '{ print $2 - 5 }' file.txt
    Subtract 5 from the second field.

awk '{ print $2 * 2 }' file.txt
    Multiply the second field by 2.

awk '{ print $2 / 2 }' file.txt
    Divide the second field by 2.

awk '{ print $2 % 3 }' file.txt
    Modulo: print the remainder of the second field divided by 3.

awk '{ x = $2 * 2; print x }' file.txt
    Assign an arithmetic expression to a variable and print it.

awk '{ print -$2 }' file.txt
    Unary minus: negate the second field.

# --- Compound assignment operators ---

awk '{ x = 10; x += 5; print x }' file.txt
    Add 5 to x and print the result (15).

awk '{ x = 10; x *= 3; print x }' file.txt
    Multiply x by 3 and print the result (30).

awk '{ x = 10; x -= 2; print x }' file.txt
    Subtract 2 from x and print the result (8).

awk '{ x = 10; x /= 2; print x }' file.txt
    Divide x by 2 and print the result (5).

awk '{ x = 10; x %= 3; print x }' file.txt
    Assign the remainder of x / 3 to x and print (1).

# --- Field separator ---

awk -F, '{ print $1 }' file.txt
    Use comma as the field separator via the `-F` flag.

awk -F: '{ print $1, $2 }' file.txt
    Use colon as the field separator.

awk 'BEGIN { FS = "," } { print $1 }' file.txt
    Set the field separator by assigning the FS variable in a BEGIN block.

awk 'BEGIN { FS = ":" } { print $2 }' file.txt
    Set the field separator to colon via the FS variable.

awk '{ print FS }' file.txt
    Print the current field separator (default is a single space " ").

awk 'BEGIN { FS = " " } { print $1, $2 }' file.txt
    Default behavior: a single space as FS splits on any sequence of whitespace.

awk 'BEGIN { FS = "" } { print $1 }' file.txt
    Empty FS: split each character into its own field.

# --- Reading scripts from files (-f flag) ---

awk -f script.awk file.txt
    Read the AWK program from script.awk instead of the command line.

awk -f script.awk
    Read the program from a file and process stdin.

# --- Setting variables with -v ---

awk -v name=John '{ print name }' file.txt
    Set variable `name` to "John" before execution. Accessible in all blocks.

awk -v OFS=',' '{ print $1, $2 }' file.txt
    Set built-in variable OFS (Output Field Separator) via -v.

awk -v count=0 -v prefix=">>" '{ print prefix, $0 }' file.txt
    Multiple -v flags can be used.

# --- Comments ---

awk '# this is a comment
BEGIN { print "hello" }'
    Lines starting with # are comments, ignored by the interpreter.

# --- if/else ---

awk '{ if ($1 > 10) print $1, "big" }' file.txt
    Print the first field and "big" if it's greater than 10.

awk '{ if ($1 > 10) print $1, "big"; else print $1, "small" }' file.txt
    Print "big" if field 1 > 10, otherwise print "small".

awk '{
    if ($1 >= 80) print $1, "A"
    else if ($1 >= 60) print $1, "B"
    else print $1, "C"
}' file.txt
    if-else if-else chain for grading.

awk '{ if ($1 % 2 == 0) print "even"; else print "odd" }' file.txt
    Check even/odd using the modulo operator.

awk '{
    if ($1 > 50 && $2 < 10) print $1, "special"
    else print $1, "normal"
}' file.txt
    Compound condition with && (logical AND).

awk '{
    if ($1 ~ /error/) print $1, "error found"
    else print $1, "no error"
}' file.txt
    Match a regex pattern inside an if condition.

awk '{
    if ($1 > max) {
        max = $1
        topper = $2
    }
}
END { print "Max:", max, "Name:", topper }' file.txt
    Nested if block with multiple statements inside braces.

awk 'BEGIN { x = 10; if (x > 5) print "yes"; else print "no" }'
    if/else inside a BEGIN block, no input file needed.

# --- while loops ---

awk 'BEGIN { i = 0; while (i < 5) { print i; i++ } }'
    While loop: print 0 through 4.

awk '{
    total = 0
    i = 1
    while (i <= NF) {
        total += $i
        i++
    }
    print "sum:", total
}' file.txt
    While loop to sum all fields on each line.

# --- for loops ---

awk 'BEGIN { for (i = 0; i < 5; i++) print i }'
    C-style for loop: print 0 through 4.

awk '{
    sum = 0
    for (i = 1; i <= NF; i++) sum += $i
    print "sum:", sum
}' file.txt
    For loop to sum all fields on each line.

awk '{
    for (i = NF; i >= 1; i--) printf "%s ", $i
    print ""
}' file.txt
    For loop printing fields in reverse order.

# --- do-while loops ---

awk 'BEGIN {
    i = 0
    do {
        print i
        i++
    } while (i < 5)
}'
    Do-while loop: executes the body at least once, prints 0 through 4.

# --- break ---

awk 'BEGIN {
    for (i = 0; i < 10; i++) {
        if (i == 5) break
        print i
    }
}'
    Break: exit the loop when i reaches 5 (prints 0-4).

awk 'BEGIN {
    i = 0
    while (1) {
        if (i >= 3) break
        print i
        i++
    }
}'
    Break with while: exit an infinite loop.

awk '{
    for (i = 1; i <= NF; i++) {
        if ($i == "") break
        print $i
    }
}' file.txt
    Break out of a loop when encountering an empty field.

# --- continue ---

awk 'BEGIN {
    for (i = 0; i < 10; i++) {
        if (i % 2 == 0) continue
        print i
    }
}'
    Continue: skip even numbers, print only odd numbers (1, 3, 5, 7, 9).

# --- next ---

awk '{ if ($1 == "") next; print $1 }' file.txt
    Next: skip the current record and move to the next input line.

awk '/^#/ { next } { print }' file.txt
    Skip comment lines starting with #.

# --- exit ---

awk 'BEGIN { print "start"; exit; print "never" }'
    Exit immediately in a BEGIN block; "never" is not printed.

awk '{ if (NR > 5) exit; print NR, $1 }' file.txt
    Exit after processing 5 records.

awk '{ if (NR > 3) exit 99; print NR }' file.txt
    Exit with a specific exit code (99).

awk 'BEGIN { exit 1 } END { print "END" }'
    Exit in BEGIN skips input processing but still runs END.

# --- Nested loops ---

awk 'BEGIN {
    for (i = 1; i <= 3; i++) {
        for (j = 1; j <= 3; j++) {
            printf "%d-%d ", i, j
        }
        print ""
    }
}'
    Nested for loops: prints a 3x3 grid of pairs.

awk 'BEGIN {
    i = 1
    while (i <= 3) {
        j = 1
        while (j <= 3) {
            print i, j
            j++
        }
        i++
    }
}'
    Nested while loops: prints a 3x3 grid.

# --- Loop with if/break/continue ---

awk '{
    for (i = 1; i <= NF; i++) {
        if ($i < 0) continue
        if ($i > 100) break
        print $i
    }
}' file.txt
    Skip negative values (continue) and stop at values over 100 (break).
